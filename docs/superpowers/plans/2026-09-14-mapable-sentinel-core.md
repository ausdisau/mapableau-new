# MapAble Sentinel Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the shared MapAble Sentinel contracts, deterministic policy adapter, feature flags and Temporal-backed supervisory workflow without changing any Care, Transport or Employment production behaviour.

**Architecture:** Sentinel is additive beneath existing domain services. It reuses the Unified Guardian as deterministic policy authority, stores only orchestration references/state, uses Temporal for durable waiting/retry, and leaves execution to the Governed Action Kernel.

**Tech Stack:** Next.js 15.5.21, TypeScript, Zod 4.3.6, Vitest 3.2.7, pnpm 10.12.1, Temporal TypeScript SDK 1.23.0 (`@temporalio/client`, `@temporalio/worker`, `@temporalio/workflow`, `@temporalio/activity`, `@temporalio/testing`).

**Spec:** `docs/superpowers/specs/2026-09-14-mapable-sentinel-runtime-transport-employment-design.md`

## Global Constraints

- Autonomy ceiling is **B — Guardian Sentinel**.
- Existing `lib/ai/platform/guardian/` remains deterministic policy authority.
- Existing `lib/ai/platform/actions/` remains execution boundary.
- Models may signal/explain but never authorise, execute, determine reportability, substantiate abuse, approve funding/payment, make employment selection decisions, call emergency services or actuate devices.
- Sentinel stores orchestration references/state, not duplicate participant/domain records.
- Unknown evidence remains unknown.
- All `MAPABLE_SENTINEL_*` flags default `false`.
- No production gating is enabled by this plan.
- Temporal Workflow code contains no database/network/clock side effects outside Temporal-safe APIs; those belong in Activities.
- All `@temporalio/*` packages use exactly `1.23.0`.
- WCAG 2.2 AA and non-AI/manual fallbacks remain mandatory for later UI work.

```ts
  to: SentinelState,
): boolean {
  return TRANSITIONS[from].includes(to);

  from: SentinelState,
  to: SentinelState,
```

```ts
import type { GuardianDecision } from "@/lib/ai/platform/guardian";

import type {
  SentinelPolicyDecision,
  SentinelPolicyDisposition,
} from "./contracts";

function dispositionFor(
  decision: GuardianDecision["decision"],
): SentinelPolicyDisposition {
  switch (decision) {
    case "ALLOW":
      return "ALLOW";
    case "ALLOW_WITH_CONDITIONS":
    case "REDACT_BEFORE_PROCESSING":
      return "ALLOW_WITH_CONDITIONS";
    case "REQUIRE_PARTICIPANT_CONFIRMATION":
      return "PARTICIPANT_CONFIRMATION";
    case "ROUTE_TO_HUMAN_REVIEW":
      return "HUMAN_REVIEW";
    case "ROUTE_TO_COMPLAINTS":
      return "COMPLAINT_HANDOFF";
    case "ROUTE_TO_INCIDENT_TRIAGE":
      return "INCIDENT_TRIAGE_HANDOFF";
    case "DENY_DATA_DISCLOSURE":
    case "DENY_PROHIBITED_ACTION":
      return "DENY";
    case "SECURITY_QUARANTINE":
      return "SECURITY_QUARANTINE";
    default: {
      const exhaustive: never = decision;
      return exhaustive;
    }
  }
}

export function mapGuardianDecisionToSentinel(
  decision: GuardianDecision,
): SentinelPolicyDecision {
  return {
    disposition: dispositionFor(decision.decision),
    reasonCodes: [...decision.reasonCodes],
    policyVersion: decision.policyVersion,
    participantConfirmationRequired:
      decision.participantConfirmationRequired,
    humanReviewRequired: decision.humanReviewRequired,
  };
}
```

```ts
// lib/ai/platform/sentinel/temporal/workflows/sentinel-case.workflow.ts
import {
  condition,
  defineQuery,
  defineSignal,
  setHandler,
} from "@temporalio/workflow";

import { assertSentinelTransition } from "../../state-machine";
import type {
  SentinelWorkflowInput,
  SentinelWorkflowState,
} from "../contracts";

export const participantStopSignal = defineSignal("participantStop");
export const participantConfirmSignal = defineSignal("participantConfirm");
export const humanReviewSignal = defineSignal<["approved" | "rejected"]>(
  "humanReview",
);
export const sentinelStateQuery = defineQuery<SentinelWorkflowState["state"]>(
  "sentinelState",
);

export async function sentinelCaseWorkflow(
  input: SentinelWorkflowInput,
): Promise<SentinelWorkflowState> {
  let state: SentinelWorkflowState = { ...input, state: input.initialState };
  let finished = false;

  setHandler(sentinelStateQuery, () => state.state);
  setHandler(participantStopSignal, () => {
    assertSentinelTransition(state.state, "STOPPED_BY_PARTICIPANT");
    state = {
      ...state,
      state: "STOPPED_BY_PARTICIPANT",
      participantDecision: "stopped",
    };
    finished = true;
  });
  setHandler(participantConfirmSignal, () => {

    state = {
      ...state,
      state: "APPROVED_ACTION_PENDING",
      participantDecision: "confirmed",
    };
    finished = true;
  });
  setHandler(humanReviewSignal, (decision) => {
    const next = decision === "approved" ? "APPROVED_ACTION_PENDING" : "FAILED_SAFE";
    assertSentinelTransition(state.state, next);
    state = { ...state, state: next, humanReviewDecision: decision };
    finished = true;
  });

  await condition(() => finished);
  return state;
}
```

```ts
import { NativeConnection, Worker } from "@temporalio/worker";

import { isSentinelTemporalAllowed } from "@/lib/config/sentinel";

async function main() {
  if (!isSentinelTemporalAllowed()) {

    address: process.env.TEMPORAL_ADDRESS ?? "localhost:7233",

    namespace: process.env.TEMPORAL_NAMESPACE ?? "default",

}

main().catch((error) => {
```
